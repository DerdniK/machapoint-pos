Crear `carrito.js` Este archivo únicamente recuperará lo que guardamos en `localStorage`. ```javascript const CARRITO_KEY = 'carrito'; const listaCarrito = document.getElementById('lista-carrito'); const mensajeCarrito = document.getElementById('mensaje-carrito'); function cargarCarrito() { const carrito = JSON.parse( localStorage.getItem(CARRITO_KEY) ) || []; console.log( "Carrito recuperado:", carrito ); if (carrito.length === 0) { mensajeCarrito.textContent = "El carrito está vacío."; return; } mensajeCarrito.textContent = ''; carrito.forEach(producto => { const article = document.createElement('article'); article.classList.add( 'producto-carrito' ); article.innerHTML = `

${producto.name}

SKU: ${producto.sku}

Tipo: ${producto.typeid}

$${Number(producto.price).toFixed(2)}
`; listaCarrito.appendChild(article); }); } // Ejecutar al cargar cargarCarrito(); ``` --- # 6.
